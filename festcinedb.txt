/* ============================================================
   FESTCINE — Sistema de Gestión de Festival de Cine
   Fase 1 y 2: DDL (CREATE + ALTER TABLE) y DML
   Motor: PostgreSQL 18
   Autor: Javier Torrico Medina — 2026
   ============================================================

   ASUNCIONES DOCUMENTADAS:
   1. Un asistente VIP tiene tarifa $0 pero igual se registra
      la entrada para control de aforo.
   2. La "pasarela de pago" en la venta de abonos se simula
      con un parámetro booleano p_PagoExitoso. Si es FALSE
      se ejecuta el ROLLBACK.
   3. Los abonos no especifican proyecciones concretas;
      dan acceso general según su tipo (fin de semana, total).
   4. El histórico de ediciones se gestiona con la tabla
      Ediciones; las proyecciones y ventas de cada edición
      se identifican por la fecha de la proyección.
   5. Las habitaciones de hotel se modelan como texto simple
      (no se gestiona inventario de habitaciones).
   6. La política de reembolsos queda fuera del alcance del
      sistema en esta versión.
   ============================================================ */

/* ─── Eliminar y recrear la base ──────────────────────────── */
DROP DATABASE IF EXISTS FestCine;
CREATE DATABASE FestCine;

/* Conectar a la base antes de continuar:
   En psql:    \c festcine
   En pgAdmin: seleccionar la base FestCine y abrir Query Tool */

/* ============================================================
   A. CATÁLOGO CINEMATOGRÁFICO Y PERSONAL
   ============================================================ */

CREATE TABLE Generos
(
    IdGenero        SERIAL,
    NombreGenero    VARCHAR(30)  NOT NULL,
    CONSTRAINT PK_Gen    PRIMARY KEY (IdGenero),
    CONSTRAINT UQ_GenNom UNIQUE      (NombreGenero)
);

CREATE TABLE Peliculas
(
    IdPelicula      SERIAL,
    Titulo          VARCHAR(150) NOT NULL,
    AnioProd        INT          NOT NULL CHECK (AnioProd > 1888),
    Duracion        INT          NOT NULL CHECK (Duracion > 0),
    PaisOrigen      VARCHAR(60)  NOT NULL,
    Sinopsis        TEXT,
    Clasificacion   VARCHAR(10)  NOT NULL
                        CHECK (Clasificacion IN ('G','PG','PG-13','R','NC-17','ATP')),
    Formato         VARCHAR(10)  NOT NULL
                        CHECK (Formato IN ('Digital','35mm','IMAX')),
    Estado          VARCHAR(15)  NOT NULL DEFAULT 'Postulada'
                        CHECK (Estado IN ('Postulada','Seleccionada','Rechazada','Premiada')),
    CONSTRAINT PK_Pel PRIMARY KEY (IdPelicula)
);

/* Relación N:M Pelicula ↔ Genero */
CREATE TABLE PeliculaGenero
(
    IdPelicula  INT NOT NULL,
    IdGenero    INT NOT NULL,
    CONSTRAINT PK_PelGen     PRIMARY KEY (IdPelicula, IdGenero),
    CONSTRAINT FK_PelGen_Pel FOREIGN KEY (IdPelicula) REFERENCES Peliculas,
    CONSTRAINT FK_PelGen_Gen FOREIGN KEY (IdGenero)   REFERENCES Generos
);

/* Personal centralizado: directores, actores, guionistas, productores */
CREATE TABLE Personal
(
    IdPersonal  SERIAL,
    Nombre      VARCHAR(100) NOT NULL,
    Biografia   TEXT,
    Email       VARCHAR(100),
    Telefono    VARCHAR(20),
    CONSTRAINT PK_Per PRIMARY KEY (IdPersonal)
);

/* Una persona puede tener múltiples roles en múltiples películas */
CREATE TABLE RolesPelicula
(
    IdPersonal  INT         NOT NULL,
    IdPelicula  INT         NOT NULL,
    Rol         VARCHAR(20) NOT NULL
                    CHECK (Rol IN ('Director','Actor','Guionista','Productor')),
    CONSTRAINT PK_RolPel     PRIMARY KEY (IdPersonal, IdPelicula, Rol),
    CONSTRAINT FK_RolPel_Per FOREIGN KEY (IdPersonal) REFERENCES Personal,
    CONSTRAINT FK_RolPel_Pel FOREIGN KEY (IdPelicula) REFERENCES Peliculas
);

/* ============================================================
   B. SEDES, SALAS, PROYECCIONES Y EVENTOS PARALELOS
   ============================================================ */

CREATE TABLE Sedes
(
    IdSede      SERIAL,
    NombreSede  VARCHAR(100) NOT NULL,
    Direccion   VARCHAR(200),
    Ciudad      VARCHAR(60),
    CONSTRAINT PK_Sed PRIMARY KEY (IdSede)
);

CREATE TABLE Salas
(
    IdSala      SERIAL,
    NombreSala  VARCHAR(60) NOT NULL,
    Capacidad   INT         NOT NULL CHECK (Capacidad > 0),
    IdSede      INT         NOT NULL,
    CONSTRAINT PK_Sal     PRIMARY KEY (IdSala),
    CONSTRAINT FK_Sal_Sed FOREIGN KEY (IdSede) REFERENCES Sedes
);

CREATE TABLE Proyecciones
(
    IdProyeccion    SERIAL,
    IdPelicula      INT       NOT NULL,
    IdSala          INT       NOT NULL,
    FechaHora       TIMESTAMP NOT NULL,
    TieneQA         BOOLEAN   DEFAULT FALSE,
    AforoDisponible INT       NOT NULL CHECK (AforoDisponible >= 0),
    CONSTRAINT PK_Proy     PRIMARY KEY (IdProyeccion),
    CONSTRAINT FK_Proy_Pel FOREIGN KEY (IdPelicula) REFERENCES Peliculas,
    CONSTRAINT FK_Proy_Sal FOREIGN KEY (IdSala)     REFERENCES Salas
);

CREATE TABLE EventosParalelos
(
    IdEvento         SERIAL,
    NombreEvento     VARCHAR(150) NOT NULL,
    TipoEvento       VARCHAR(15)  NOT NULL
                         CHECK (TipoEvento IN ('Masterclass','Taller','Coctel')),
    FechaHora        TIMESTAMP    NOT NULL,
    Aforo            INT          NOT NULL CHECK (Aforo > 0),
    CostoInscripcion NUMERIC(10,2) DEFAULT 0 CHECK (CostoInscripcion >= 0),
    CONSTRAINT PK_Evt PRIMARY KEY (IdEvento)
);

/* Expositores invitados a eventos (usa tabla Personal) */
CREATE TABLE ExpositorEvento
(
    IdEvento    INT NOT NULL,
    IdPersonal  INT NOT NULL,
    CONSTRAINT PK_ExpEvt     PRIMARY KEY (IdEvento, IdPersonal),
    CONSTRAINT FK_ExpEvt_Evt FOREIGN KEY (IdEvento)   REFERENCES EventosParalelos,
    CONSTRAINT FK_ExpEvt_Per FOREIGN KEY (IdPersonal) REFERENCES Personal
);

/* ============================================================
   C. COMPETICIÓN, JURADOS Y PREMIOS
   ============================================================ */

CREATE TABLE Categorias
(
    IdCategoria     SERIAL,
    NombreCategoria VARCHAR(80)  NOT NULL,
    Descripcion     VARCHAR(200),
    CONSTRAINT PK_Cat PRIMARY KEY (IdCategoria)
);

CREATE TABLE MiembrosJurado
(
    IdMiembro   SERIAL,
    Nombre      VARCHAR(100) NOT NULL,
    Profesion   VARCHAR(60),
    Pais        VARCHAR(60),
    Email       VARCHAR(100),
    CONSTRAINT PK_MJ PRIMARY KEY (IdMiembro)
);

/* Un miembro puede estar en varias categorías */
CREATE TABLE JuradoCategoria
(
    IdMiembro   INT NOT NULL,
    IdCategoria INT NOT NULL,
    CONSTRAINT PK_JurCat     PRIMARY KEY (IdMiembro, IdCategoria),
    CONSTRAINT FK_JurCat_MJ  FOREIGN KEY (IdMiembro)   REFERENCES MiembrosJurado,
    CONSTRAINT FK_JurCat_Cat FOREIGN KEY (IdCategoria) REFERENCES Categorias
);

/* Películas que compiten en cada categoría */
CREATE TABLE CompetenciaPelicula
(
    IdPelicula  INT NOT NULL,
    IdCategoria INT NOT NULL,
    CONSTRAINT PK_CompPel     PRIMARY KEY (IdPelicula, IdCategoria),
    CONSTRAINT FK_CompPel_Pel FOREIGN KEY (IdPelicula)  REFERENCES Peliculas,
    CONSTRAINT FK_CompPel_Cat FOREIGN KEY (IdCategoria) REFERENCES Categorias
);

/* Evaluaciones del jurado: puntuación 1-10 y comentario */
CREATE TABLE Evaluaciones
(
    IdEvaluacion    SERIAL,
    IdMiembro       INT  NOT NULL,
    IdPelicula      INT  NOT NULL,
    IdCategoria     INT  NOT NULL,
    Puntuacion      INT  NOT NULL CHECK (Puntuacion BETWEEN 1 AND 10),
    Comentario      TEXT,
    CONSTRAINT PK_Eval    PRIMARY KEY (IdEvaluacion),
    CONSTRAINT UQ_Eval    UNIQUE      (IdMiembro, IdPelicula, IdCategoria),
    CONSTRAINT FK_Eval_JC FOREIGN KEY (IdMiembro, IdCategoria)
                              REFERENCES JuradoCategoria (IdMiembro, IdCategoria),
    CONSTRAINT FK_Eval_CP FOREIGN KEY (IdPelicula, IdCategoria)
                              REFERENCES CompetenciaPelicula (IdPelicula, IdCategoria)
);

/* Película ganadora por categoría y edición */
CREATE TABLE Premios
(
    IdPremio    SERIAL,
    IdCategoria INT NOT NULL,
    IdPelicula  INT NOT NULL,
    AnioEdicion INT NOT NULL DEFAULT EXTRACT(YEAR FROM CURRENT_DATE),
    CONSTRAINT PK_Pre       PRIMARY KEY (IdPremio),
    CONSTRAINT UQ_PreCatAno UNIQUE      (IdCategoria, AnioEdicion),
    CONSTRAINT FK_Pre_Cat   FOREIGN KEY (IdCategoria) REFERENCES Categorias,
    CONSTRAINT FK_Pre_Pel   FOREIGN KEY (IdPelicula)  REFERENCES Peliculas
);

/* ============================================================
   D. ASISTENTES, TARIFAS, ENTRADAS Y ABONOS
   ============================================================ */

CREATE TABLE Asistentes
(
    IdAsistente     SERIAL,
    Nombre          VARCHAR(100) NOT NULL,
    Email           VARCHAR(100) NOT NULL,
    Telefono        VARCHAR(20),
    TipoAsistente   VARCHAR(15)  NOT NULL DEFAULT 'General'
                        CHECK (TipoAsistente IN ('General','Prensa','Industria','VIP','Jurado')),
    CONSTRAINT PK_Asi      PRIMARY KEY (IdAsistente),
    CONSTRAINT UQ_AsiEmail UNIQUE      (Email)
);

CREATE TABLE Tarifas
(
    IdTarifa    SERIAL,
    NombreTarifa VARCHAR(30)   NOT NULL,
    Precio       NUMERIC(10,2) NOT NULL CHECK (Precio >= 0),
    CONSTRAINT PK_Tar PRIMARY KEY (IdTarifa)
);

/* Entradas individuales: válidas para una proyección O un evento paralelo.
   Asunción documentada: IdProyeccion e IdEvento son mutuamente excluyentes;
   exactamente uno debe tener valor y el otro NULL.
   El CHECK garantiza esta exclusividad. */
CREATE TABLE Entradas
(
    IdEntrada    SERIAL,
    IdAsistente  INT       NOT NULL,
    IdProyeccion INT,
    IdEvento     INT,
    IdTarifa     INT       NOT NULL,
    FechaCompra  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT PK_Ent          PRIMARY KEY (IdEntrada),
    CONSTRAINT CK_Ent_Destino  CHECK (
                                   (IdProyeccion IS NOT NULL AND IdEvento IS NULL) OR
                                   (IdProyeccion IS NULL     AND IdEvento IS NOT NULL)
                               ),
    /* UNIQUE parcial: PostgreSQL ignora NULLs en UNIQUE convencional.
       Se usan índices parciales para garantizar unicidad real por caso. */
    CONSTRAINT FK_Ent_Asi      FOREIGN KEY (IdAsistente)  REFERENCES Asistentes,
    CONSTRAINT FK_Ent_Proy     FOREIGN KEY (IdProyeccion) REFERENCES Proyecciones,
    CONSTRAINT FK_Ent_Evt      FOREIGN KEY (IdEvento)     REFERENCES EventosParalelos,
    CONSTRAINT FK_Ent_Tar      FOREIGN KEY (IdTarifa)     REFERENCES Tarifas
);

/* Índices parciales para unicidad real con columnas NULLables.
   Un asistente no puede tener dos entradas a la misma proyección,
   ni dos entradas al mismo evento. */
CREATE UNIQUE INDEX UX_Ent_AsiProy
    ON Entradas (IdAsistente, IdProyeccion)
    WHERE IdProyeccion IS NOT NULL;

CREATE UNIQUE INDEX UX_Ent_AsiEvt
    ON Entradas (IdAsistente, IdEvento)
    WHERE IdEvento IS NOT NULL;

CREATE TABLE TiposAbono
(
    IdTipoAbono SERIAL,
    NombreAbono VARCHAR(60)   NOT NULL,
    Descripcion VARCHAR(200),
    Precio      NUMERIC(10,2) NOT NULL CHECK (Precio >= 0),
    CONSTRAINT PK_TipoAb PRIMARY KEY (IdTipoAbono)
);

/* Abonos: pases de acceso múltiple */
CREATE TABLE Abonos
(
    IdAbono     SERIAL,
    IdAsistente INT         NOT NULL,
    IdTipoAbono INT         NOT NULL,
    FechaCompra TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CodigoAcceso VARCHAR(20) NOT NULL,
    Pagado      BOOLEAN     NOT NULL DEFAULT FALSE,
    CONSTRAINT PK_Ab      PRIMARY KEY (IdAbono),
    CONSTRAINT UQ_AbCod   UNIQUE      (CodigoAcceso),
    CONSTRAINT FK_Ab_Asi  FOREIGN KEY (IdAsistente) REFERENCES Asistentes,
    CONSTRAINT FK_Ab_Tipo FOREIGN KEY (IdTipoAbono) REFERENCES TiposAbono
);

/* ============================================================
   E. LOGÍSTICA Y PATROCINIOS (HISTÓRICOS)
   ============================================================ */

CREATE TABLE Hoteles
(
    IdHotel     SERIAL,
    NombreHotel VARCHAR(100) NOT NULL,
    Direccion   VARCHAR(200),
    Estrellas   INT CHECK (Estrellas BETWEEN 1 AND 5),
    CONSTRAINT PK_Hot PRIMARY KEY (IdHotel)
);

CREATE TABLE Alojamientos
(
    IdAlojamiento   SERIAL,
    IdPersonal      INT         NOT NULL,
    IdHotel         INT         NOT NULL,
    NroHabitacion   VARCHAR(10) NOT NULL,
    CheckIn         DATE        NOT NULL,
    CheckOut        DATE        NOT NULL,
    CONSTRAINT PK_Aloj        PRIMARY KEY (IdAlojamiento),
    CONSTRAINT CK_Aloj_Fechas CHECK       (CheckOut > CheckIn),
    CONSTRAINT FK_Aloj_Per    FOREIGN KEY (IdPersonal) REFERENCES Personal,
    CONSTRAINT FK_Aloj_Hot    FOREIGN KEY (IdHotel)    REFERENCES Hoteles
);

CREATE TABLE Traslados
(
    IdTraslado  SERIAL,
    IdPersonal  INT         NOT NULL,
    TipoTraslado VARCHAR(10) NOT NULL CHECK (TipoTraslado IN ('Vuelo','Transfer')),
    Origen      VARCHAR(100) NOT NULL,
    Destino     VARCHAR(100) NOT NULL,
    FechaHora   TIMESTAMP    NOT NULL,
    NroVuelo    VARCHAR(20),
    CONSTRAINT PK_Tras     PRIMARY KEY (IdTraslado),
    CONSTRAINT FK_Tras_Per FOREIGN KEY (IdPersonal) REFERENCES Personal
);

CREATE TABLE Ediciones
(
    IdEdicion   SERIAL,
    Anio        INT          NOT NULL,
    NombreEdicion VARCHAR(100),
    FechaInicio DATE         NOT NULL,
    FechaFin    DATE         NOT NULL,
    CONSTRAINT PK_Edi        PRIMARY KEY (IdEdicion),
    CONSTRAINT UQ_EdiAnio    UNIQUE      (Anio),
    CONSTRAINT CK_Edi_Fechas CHECK       (FechaFin > FechaInicio)
);

CREATE TABLE Patrocinadores
(
    IdPatrocinador SERIAL,
    NombreEmpresa  VARCHAR(100) NOT NULL,
    Contacto       VARCHAR(100),
    Email          VARCHAR(100),
    CONSTRAINT PK_Patr PRIMARY KEY (IdPatrocinador)
);

CREATE TABLE PatrocinioEdicion
(
    IdPatrocinio    SERIAL,
    IdPatrocinador  INT          NOT NULL,
    IdEdicion       INT          NOT NULL,
    TipoAporte      VARCHAR(10)  NOT NULL CHECK (TipoAporte IN ('Economico','Especie')),
    Monto           NUMERIC(12,2),
    DescripcionAporte VARCHAR(200),
    CONSTRAINT PK_PatEdi     PRIMARY KEY (IdPatrocinio),
    CONSTRAINT FK_PatEdi_Pat FOREIGN KEY (IdPatrocinador) REFERENCES Patrocinadores,
    CONSTRAINT FK_PatEdi_Edi FOREIGN KEY (IdEdicion)      REFERENCES Ediciones
);

CREATE OR REPLACE VIEW vw_Peliculas AS
    SELECT p.IdPelicula,
           p.Titulo,
           p.AnioProd,
           p.Duracion,
           p.PaisOrigen,
           p.Clasificacion,
           p.Formato,
           p.Estado,
           COALESCE(STRING_AGG(g.NombreGenero, ', '), '') AS Generos
      FROM Peliculas p
      LEFT JOIN PeliculaGenero pg ON pg.IdPelicula = p.IdPelicula
      LEFT JOIN Generos g         ON g.IdGenero    = pg.IdGenero
     WHERE p.Estado IN ('Seleccionada', 'Premiada')
     GROUP BY p.IdPelicula, p.Titulo, p.AnioProd, p.Duracion,
              p.PaisOrigen, p.Clasificacion, p.Formato, p.Estado
     ORDER BY p.Titulo;

      INNER JOIN Salas sa     ON sa.IdSala     = pr.IdSala
      INNER JOIN Sedes se     ON se.IdSede     = sa.IdSede

/* ============================================================
   CONSULTA 1: Ranking de Películas
   Películas más vistas, con porcentaje de ocupación de salas.
   Asistentes reales vs. Capacidad de la sala.
   ============================================================ */
SELECT
    p.Titulo,
    COUNT(e.IdEntrada)                                              AS TotalAsistentes,
    SUM(s.Capacidad)                                               AS CapacidadTotal,
    ROUND(
        COUNT(e.IdEntrada) * 100.0 / NULLIF(SUM(s.Capacidad), 0),
    2)                                                             AS PorcentajeOcupacion
FROM Peliculas p
    INNER JOIN Proyecciones pr ON pr.IdPelicula  = p.IdPelicula
    INNER JOIN Salas s         ON s.IdSala        = pr.IdSala
    LEFT  JOIN Entradas e      ON e.IdProyeccion  = pr.IdProyeccion
GROUP BY p.IdPelicula, p.Titulo
ORDER BY TotalAsistentes DESC, PorcentajeOcupacion DESC;

   CONSULTA 2: Acta de Premiación
   Películas ganadoras por categoría con promedio de votación.
   ============================================================ */

    c.NombreCategoria,
    p.Titulo                      AS PeliculaGanadora,
    ROUND(AVG(ev.Puntuacion), 2)  AS PromedioJurado,
    COUNT(ev.IdEvaluacion)        AS VotosRecibidos,
    pr.AnioEdicion
FROM Premios pr
    INNER JOIN Categorias c    ON c.IdCategoria  = pr.IdCategoria
    INNER JOIN Peliculas p     ON p.IdPelicula   = pr.IdPelicula
    INNER JOIN Evaluaciones ev ON ev.IdPelicula  = pr.IdPelicula
                              AND ev.IdCategoria = pr.IdCategoria
GROUP BY c.NombreCategoria, p.Titulo, pr.AnioEdicion
ORDER BY pr.AnioEdicion, c.NombreCategoria;

/* ============================================================
   P1: Procedimiento Almacenado — "Proceso de Compra de Entrada"
   Parámetros IN:  p_IdAsistente, p_IdProyeccion, p_IdTarifa
   Parámetro OUT:  p_Respuesta (resultado de la operación)
   Lógica: verifica aforo → inserta entrada → reduce aforo
           si no hay aforo, lanza error descriptivo
   ============================================================ */
CREATE OR REPLACE PROCEDURE ComprarEntrada(
    IN  p_IdAsistente  INT,
    IN  p_IdProyeccion INT,
    IN  p_IdTarifa     INT,
    OUT p_Respuesta    VARCHAR(300)
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_Aforo     INT;
    v_Titulo    VARCHAR(150);
    v_FechaHora TIMESTAMP;
    v_YaCompro  INT;
BEGIN
    /* 1. Verificar que la proyección exista */
    SELECT pr.AforoDisponible, pe.Titulo, pr.FechaHora
        INTO v_Aforo, v_Titulo, v_FechaHora
        FROM Proyecciones pr
            INNER JOIN Peliculas pe ON pe.IdPelicula = pr.IdPelicula
        WHERE pr.IdProyeccion = p_IdProyeccion;

    IF NOT FOUND THEN
        p_Respuesta := 'Error: La proyeccion indicada no existe.';
        RETURN;
    END IF;

    /* 2. Verificar que el asistente no tenga ya una entrada */
    SELECT COUNT(*) INTO v_YaCompro
        FROM Entradas
        WHERE IdAsistente  = p_IdAsistente
          AND IdProyeccion = p_IdProyeccion;

    IF v_YaCompro > 0 THEN
        p_Respuesta := 'Error: El asistente ya tiene una entrada para esta proyeccion.';
        RETURN;
    END IF;

    /* 3. Verificar aforo disponible */
    IF v_Aforo <= 0 THEN
        p_Respuesta := 'Lo sentimos, no hay aforo disponible para esta funcion.';
        RETURN;
    END IF;

    /* 4. Registrar la entrada */
    INSERT INTO Entradas (IdAsistente, IdProyeccion, IdTarifa)
        VALUES (p_IdAsistente, p_IdProyeccion, p_IdTarifa);

    /* 5. Reducir el aforo disponible */
    UPDATE Proyecciones
        SET AforoDisponible = AforoDisponible - 1
        WHERE IdProyeccion = p_IdProyeccion;

    p_Respuesta := 'Entrada registrada exitosamente para "' || v_Titulo ||
                   '" el ' || TO_CHAR(v_FechaHora, 'DD/MM/YYYY HH24:MI') || '.';

EXCEPTION
    WHEN OTHERS THEN
        p_Respuesta := 'Error inesperado: ' || SQLERRM;
END;
$$;

/* ============================================================
   T1: Transacción Crítica — "Venta de Abono"
   Garantiza atomicidad completa.
   Parámetros IN:  p_IdAsistente, p_IdTipoAbono, p_PagoExitoso
   Parámetro OUT:  p_Respuesta
   Lógica:
     - Registra el pago del abono
     - Genera el código de acceso del cliente
     - Emite la factura
     - Si p_PagoExitoso = FALSE (pasarela falla) → ROLLBACK
     - Si datos inconsistentes → ROLLBACK en EXCEPTION
   NOTA PostgreSQL: COMMIT/ROLLBACK en PROCEDURE solo es válido
   cuando el procedure se llama directamente (no desde DO $$).
   Las pruebas de ROLLBACK se hacen con CALL directo.
   ============================================================ */
CREATE OR REPLACE PROCEDURE VenderAbono(
    IN  p_IdAsistente  INT,
    IN  p_IdTipoAbono  INT,
    IN  p_PagoExitoso  BOOLEAN,
    OUT p_Respuesta    VARCHAR(300)
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_CodigoAcceso VARCHAR(20);
    v_NombreAbono  VARCHAR(60);
    v_Precio       NUMERIC(10,2);
    v_NombreAsist  VARCHAR(100);
    v_NroAbono     INT;
BEGIN
    /* 1. Verificar tipo de abono */
    SELECT NombreAbono, Precio
        INTO v_NombreAbono, v_Precio
        FROM TiposAbono
        WHERE IdTipoAbono = p_IdTipoAbono;

    IF NOT FOUND THEN
        p_Respuesta := 'Error: El tipo de abono indicado no existe.';
        RETURN;
    END IF;

    /* 2. Verificar asistente */
    SELECT Nombre INTO v_NombreAsist
        FROM Asistentes
        WHERE IdAsistente = p_IdAsistente;

    IF NOT FOUND THEN
        p_Respuesta := 'Error: El asistente indicado no existe.';
        RETURN;
    END IF;

    /* 3. Generar código de acceso único */
    v_CodigoAcceso := 'AB-' || TO_CHAR(CURRENT_DATE, 'YYYY') || '-' ||
                      LPAD(CAST(FLOOR(RANDOM() * 90000 + 10000) AS TEXT), 5, '0');

    /* 4. Verificar pasarela de pago */
    IF p_PagoExitoso = FALSE THEN
        RAISE NOTICE 'Pasarela de pago fallida. Aplicando ROLLBACK...';
        p_Respuesta := 'Error: Pasarela de pago fallida. Operacion cancelada (ROLLBACK).';
        ROLLBACK;
        RETURN;
    END IF;

    /* 5. Registrar el abono (pago confirmado) */
    INSERT INTO Abonos (IdAsistente, IdTipoAbono, CodigoAcceso, Pagado)
        VALUES (p_IdAsistente, p_IdTipoAbono, v_CodigoAcceso, TRUE)
        RETURNING IdAbono INTO v_NroAbono;

    /* 6. Emitir factura */
    RAISE NOTICE '=========================================';
    RAISE NOTICE 'FACTURA NRO : F-2026-%', LPAD(v_NroAbono::TEXT, 6, '0');
    RAISE NOTICE 'Cliente     : %', v_NombreAsist;
    RAISE NOTICE 'Abono       : %', v_NombreAbono;
    RAISE NOTICE 'Monto       : Bs. %', v_Precio;
    RAISE NOTICE 'Cod. Acceso : %', v_CodigoAcceso;
    RAISE NOTICE 'Fecha       : %', TO_CHAR(CURRENT_TIMESTAMP, 'DD/MM/YYYY HH24:MI');
    RAISE NOTICE '=========================================';

    p_Respuesta := 'Abono registrado. Cod. acceso: ' || v_CodigoAcceso ||
                   ' | Factura F-2026-' || LPAD(v_NroAbono::TEXT, 6, '0');

    COMMIT;

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        p_Respuesta := 'Error critico: ' || SQLERRM || ' — ROLLBACK aplicado.';
END;
$$;

/* ============================================================
   TR1: Trigger — "Control de Agenda"
   BEFORE INSERT en la tabla Proyecciones.
   Impide programar una pelicula en una sala si ya está
   ocupada en ese rango de fecha/hora, considerando la
   duración de la película + 30 minutos de limpieza.
   ============================================================ */
CREATE OR REPLACE FUNCTION fn_ControlAgenda()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE
    v_Duracion    INT;
    v_FinNueva    TIMESTAMP;
    v_Conflicto   INT;
    v_TituloOcupa VARCHAR(150);
BEGIN
    /* Duración de la película a programar */
    SELECT Duracion INTO v_Duracion
        FROM Peliculas
        WHERE IdPelicula = NEW.IdPelicula;

    /* Fin de la nueva proyección: duración + 30 min limpieza */
    v_FinNueva := NEW.FechaHora + (v_Duracion + 30) * INTERVAL '1 minute';

    /* Buscar conflicto en la misma sala */
    SELECT COUNT(*), MAX(pe.Titulo)
        INTO v_Conflicto, v_TituloOcupa
        FROM Proyecciones pr
            INNER JOIN Peliculas pe ON pe.IdPelicula = pr.IdPelicula
        WHERE pr.IdSala       = NEW.IdSala
          AND pr.IdProyeccion <> COALESCE(NEW.IdProyeccion, -1)
          AND NEW.FechaHora   < (pr.FechaHora + (pe.Duracion + 30) * INTERVAL '1 minute')
          AND v_FinNueva      >  pr.FechaHora;

    IF v_Conflicto > 0 THEN
        RAISE EXCEPTION
            'Control de Agenda: La sala ya esta ocupada por "%" en ese horario (incluidos 30 min de limpieza).',
            v_TituloOcupa;
    END IF;

    RETURN NEW;
END;
$$;

CREATE OR REPLACE TRIGGER TR_ControlAgenda
    BEFORE INSERT OR UPDATE ON Proyecciones
    FOR EACH ROW
    EXECUTE FUNCTION fn_ControlAgenda();

/* Reporte 1: Ranking de películas por ocupación */
CREATE OR REPLACE PROCEDURE sp_ReporteRanking(
    OUT p_Respuesta VARCHAR(200)
)
LANGUAGE plpgsql
AS $$
DECLARE
    r RECORD;
BEGIN
    RAISE NOTICE '=== RANKING DE PELICULAS — EDICION 2026 ===';
    FOR r IN (
        SELECT p.Titulo,
               COUNT(e.IdEntrada)                                                  AS Asistentes,
               SUM(s.Capacidad)                                                    AS CapacidadTotal,
               ROUND(COUNT(e.IdEntrada) * 100.0 / NULLIF(SUM(s.Capacidad), 0), 2) AS PctOcupacion
          FROM Peliculas p
              INNER JOIN Proyecciones pr ON pr.IdPelicula  = p.IdPelicula
              INNER JOIN Salas s         ON s.IdSala        = pr.IdSala
              LEFT  JOIN Entradas e      ON e.IdProyeccion  = pr.IdProyeccion
         GROUP BY p.Titulo
         ORDER BY Asistentes DESC
    ) LOOP
        RAISE NOTICE 'Pelicula: % | Asistentes: % | Ocupacion: %%%',
            r.Titulo, r.Asistentes, r.PctOcupacion;
    END LOOP;
    p_Respuesta := 'Ranking generado exitosamente.';
EXCEPTION
    WHEN OTHERS THEN
        p_Respuesta := 'Error al generar ranking: ' || SQLERRM;
END;
$$;

/* Reporte 2: Acta de premiación */
CREATE OR REPLACE PROCEDURE sp_ReportePremiacion(
    OUT p_Respuesta VARCHAR(200)
)
LANGUAGE plpgsql
AS $$
DECLARE
    r RECORD;
BEGIN
    RAISE NOTICE '=== ACTA DE PREMIACION — EDICION 2026 ===';
    FOR r IN (
        SELECT c.NombreCategoria,
               p.Titulo                      AS PeliculaGanadora,
               ROUND(AVG(ev.Puntuacion), 2)  AS PromedioJurado,
               pre.AnioEdicion
          FROM Premios pre
              INNER JOIN Categorias c    ON c.IdCategoria   = pre.IdCategoria
              INNER JOIN Peliculas p     ON p.IdPelicula    = pre.IdPelicula
              INNER JOIN Evaluaciones ev ON ev.IdPelicula   = pre.IdPelicula
                                       AND ev.IdCategoria   = pre.IdCategoria
         GROUP BY c.NombreCategoria, p.Titulo, pre.AnioEdicion
         ORDER BY c.NombreCategoria
    ) LOOP
        RAISE NOTICE 'Categoria: % | Ganadora: % | Promedio Jurado: %',
            r.NombreCategoria, r.PeliculaGanadora, r.PromedioJurado;
    END LOOP;
    p_Respuesta := 'Acta de premiacion generada exitosamente.';
EXCEPTION
    WHEN OTHERS THEN
        p_Respuesta := 'Error al generar acta: ' || SQLERRM;
END;
$$;

/* Reporte 3: Informe financiero */
CREATE OR REPLACE PROCEDURE sp_ReporteFinanciero(
    OUT p_Respuesta VARCHAR(200)
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_TotalEntradas NUMERIC := 0;
    v_TotalAbonos   NUMERIC := 0;
    r               RECORD;
BEGIN
    RAISE NOTICE '=== INFORME FINANCIERO — EDICION 2026 ===';

    /* Detalle por tarifa */
    FOR r IN (
        SELECT t.NombreTarifa,
               COUNT(e.IdEntrada)  AS Cantidad,
               SUM(t.Precio)       AS Subtotal
          FROM Entradas e
              INNER JOIN Tarifas t ON t.IdTarifa = e.IdTarifa
         GROUP BY t.NombreTarifa
         ORDER BY Subtotal DESC
    ) LOOP
        RAISE NOTICE 'Entrada - % | Cant: % | Subtotal: Bs. %',
            r.NombreTarifa, r.Cantidad, r.Subtotal;
        v_TotalEntradas := v_TotalEntradas + r.Subtotal;
    END LOOP;

    /* Detalle por tipo de abono */
    FOR r IN (
        SELECT ta.NombreAbono,
               COUNT(a.IdAbono)    AS Cantidad,
               SUM(ta.Precio)      AS Subtotal
          FROM Abonos a
              INNER JOIN TiposAbono ta ON ta.IdTipoAbono = a.IdTipoAbono
         WHERE a.Pagado = TRUE
         GROUP BY ta.NombreAbono
         ORDER BY Subtotal DESC
    ) LOOP
        RAISE NOTICE 'Abono   - % | Cant: % | Subtotal: Bs. %',
            r.NombreAbono, r.Cantidad, r.Subtotal;
        v_TotalAbonos := v_TotalAbonos + r.Subtotal;
    END LOOP;

    RAISE NOTICE '--- RESUMEN ---';
    RAISE NOTICE 'Total Entradas : Bs. %', v_TotalEntradas;
    RAISE NOTICE 'Total Abonos   : Bs. %', v_TotalAbonos;
    RAISE NOTICE 'TOTAL GENERAL  : Bs. %', v_TotalEntradas + v_TotalAbonos;

    p_Respuesta := 'Total general: Bs. ' || (v_TotalEntradas + v_TotalAbonos)::TEXT;
EXCEPTION
    WHEN OTHERS THEN
        p_Respuesta := 'Error al generar informe: ' || SQLERRM;
END;
$$;

   Devuelven TABLE; la app las llama via SELECT * FROM fn_...()
   ============================================================ */
CREATE OR REPLACE FUNCTION fn_ReporteRanking()
RETURNS TABLE (
    Titulo              VARCHAR(150),
    TotalAsistentes     BIGINT,
    CapacidadTotal      BIGINT,
    PorcentajeOcupacion NUMERIC
)
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT p.Titulo,
           COUNT(e.IdEntrada),
           SUM(s.Capacidad),
           ROUND(COUNT(e.IdEntrada) * 100.0 / NULLIF(SUM(s.Capacidad), 0), 2)
      FROM Peliculas p
          INNER JOIN Proyecciones pr ON pr.IdPelicula = p.IdPelicula
          INNER JOIN Salas s         ON s.IdSala       = pr.IdSala
          LEFT  JOIN Entradas e      ON e.IdProyeccion = pr.IdProyeccion
     GROUP BY p.Titulo
     ORDER BY COUNT(e.IdEntrada) DESC;
END;
$$;

CREATE OR REPLACE FUNCTION fn_ReportePremiacion()
RETURNS TABLE (
    Categoria        VARCHAR(80),
    PeliculaGanadora VARCHAR(150),
    PromedioJurado   NUMERIC,
    Anio             INT
)
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT c.NombreCategoria,
           p.Titulo,
           ROUND(AVG(ev.Puntuacion), 2),
           pre.AnioEdicion
      FROM Premios pre
          INNER JOIN Categorias c    ON c.IdCategoria  = pre.IdCategoria
          INNER JOIN Peliculas p     ON p.IdPelicula   = pre.IdPelicula
          INNER JOIN Evaluaciones ev ON ev.IdPelicula  = pre.IdPelicula
                                   AND ev.IdCategoria  = pre.IdCategoria
     GROUP BY c.NombreCategoria, p.Titulo, pre.AnioEdicion
     ORDER BY c.NombreCategoria;
END;
$$;

CREATE OR REPLACE FUNCTION fn_ReporteFinanciero()
RETURNS TABLE (
    TipoVenta      TEXT,
    Detalle        VARCHAR(60),
    CantVentas     BIGINT,
    TotalRecaudado NUMERIC
)
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT 'Entrada Individual'::TEXT,
           t.NombreTarifa,
           COUNT(e.IdEntrada),
           SUM(t.Precio)
      FROM Entradas e
          INNER JOIN Tarifas t ON t.IdTarifa = e.IdTarifa
     GROUP BY t.NombreTarifa
    UNION ALL
    SELECT 'Abono'::TEXT,
           ta.NombreAbono,
           COUNT(a.IdAbono),
           SUM(ta.Precio)
      FROM Abonos a
          INNER JOIN TiposAbono ta ON ta.IdTipoAbono = a.IdTipoAbono
     WHERE a.Pagado = TRUE
     GROUP BY ta.NombreAbono
     ORDER BY TipoVenta, TotalRecaudado DESC;
END;
$$;

/* ============================================================
   PROCEDURE ADICIONAL — ProgramarProyeccion
   Para Módulo 2 de la app: elimina el INSERT embebido en Python.
   El Trigger TR_ControlAgenda actúa automáticamente en el INSERT.
   Parámetro OUT: p_Respuesta con el resultado de la operación.
   ============================================================ */
CREATE OR REPLACE PROCEDURE ProgramarProyeccion(
    IN  p_IdPelicula  INT,
    IN  p_IdSala      INT,
    IN  p_FechaHora   TIMESTAMP,
    IN  p_TieneQA     BOOLEAN,
    OUT p_IdNuevo     INT,
    OUT p_Respuesta   VARCHAR(300)
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_Capacidad INT;
BEGIN
    /* Obtener capacidad de la sala */
    SELECT Capacidad INTO v_Capacidad
        FROM Salas WHERE IdSala = p_IdSala;

    IF NOT FOUND THEN
        p_IdNuevo   := NULL;
        p_Respuesta := 'Error: La sala indicada no existe.';
        RETURN;
    END IF;

    /* INSERT — el Trigger TR_ControlAgenda valida el cruce de horario */
    INSERT INTO Proyecciones (IdPelicula, IdSala, FechaHora, TieneQA, AforoDisponible)
        VALUES (p_IdPelicula, p_IdSala, p_FechaHora, p_TieneQA, v_Capacidad)
        RETURNING IdProyeccion INTO p_IdNuevo;

    p_Respuesta := 'Proyeccion programada exitosamente. ID: ' || p_IdNuevo::TEXT;

EXCEPTION
    WHEN OTHERS THEN
        p_IdNuevo   := NULL;
        p_Respuesta := 'Error: ' || SQLERRM;
END;
$$;
